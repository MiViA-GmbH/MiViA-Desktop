export function complex(items: number[], flag: boolean): number {
	let total = 0;
	for (const item of items) {
		if (item > 1) {
			if (flag) {
				for (let i = 0; i < item; i++) {
					if (i % 2 === 0) {
						total += i;
					} else if (i % 3 === 0) {
						total -= i;
					}
				}
			}
		}
	}
	return total;
}
